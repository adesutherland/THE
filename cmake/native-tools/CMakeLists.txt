cmake_minimum_required(VERSION 3.24)
project(TheNativeTools C)
set(CMAKE_C_STANDARD 99)
get_filename_component(THE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(THE_CREXX_SOURCE "${THE_SOURCE}/../CREXX" CACHE PATH "Matching cREXX source")
set(THE_CREXX_HOST_BUILD "${THE_CREXX_SOURCE}/cmake-build-debug" CACHE PATH "Matching native SDK")
set(THE_CREXX_REVISION "8e8cfff4d6109bce277befeb9dd598027d1deb77" CACHE STRING "Qualified runtime revision")
find_package(Git REQUIRED)
execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse HEAD WORKING_DIRECTORY ${THE_CREXX_SOURCE} OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual STREQUAL THE_CREXX_REVISION)
 message(FATAL_ERROR "Use the configured cREXX revision ${THE_CREXX_REVISION}")
endif()
execute_process(COMMAND ${GIT_EXECUTABLE} diff --quiet HEAD -- CMakeLists.txt cmake interpreter platform binutils assembler compiler lib langtools rxpa third_party ports/single-threaded WORKING_DIRECTORY ${THE_CREXX_SOURCE} RESULT_VARIABLE modified)
if(NOT modified EQUAL 0)
 message(FATAL_ERROR "Use committed cREXX runtime/compiler sources")
endif()
set(CREXX_GENERATED_DIR "${THE_CREXX_HOST_BUILD}/generated" CACHE PATH "Generated headers")
file(READ "${CREXX_GENERATED_DIR}/crexx_version.h" version)
if(NOT version MATCHES "${THE_CREXX_REVISION}")
 message(FATAL_ERROR "SDK headers differ from the configured runtime revision")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CREXX_VM_CLASSIC_FULL_PROVIDERS ON CACHE BOOL "" FORCE)
add_compile_definitions(_GNU_SOURCE=1)
add_subdirectory("${THE_CREXX_SOURCE}/ports/single-threaded" vm EXCLUDE_FROM_ALL)
add_executable(the_rxbin_host "${THE_SOURCE}/src/tools_runner.c" "${THE_SOURCE}/src/wasm/crexxvm.c" "${THE_SOURCE}/src/wasm/crexxresources.c")
target_compile_definitions(the_rxbin_host PRIVATE THE_WASM_RUNNER=1)
target_link_libraries(the_rxbin_host PRIVATE rxvm_single_embed ${CMAKE_DL_LIBS})
if(NOT WIN32)
 target_link_libraries(the_rxbin_host PRIVATE m)
endif()
if(WIN32 AND CMAKE_C_COMPILER_ID STREQUAL "GNU")
 target_link_options(the_rxbin_host PRIVATE -municode)
endif()
set(THE_CLASSIC_IMAGE "" CACHE FILEPATH "Prepared closed llcbrowser image")
set(THE_CLASSIC_SHA256 "9dc18d8eba321fb258865a9a6e11301df4b721d2b7835efafa1e92fdc9cd0942" CACHE STRING "Prepared image checksum")
if(NOT EXISTS "${THE_CLASSIC_IMAGE}")
 message(FATAL_ERROR "Supply THE_CLASSIC_IMAGE")
endif()
file(SHA256 "${THE_CLASSIC_IMAGE}" checksum)
if(NOT checksum STREQUAL THE_CLASSIC_SHA256)
 message(FATAL_ERROR "Classic image checksum differs")
endif()
set(THE_LANGUAGE_IMAGE "" CACHE FILEPATH "Prepared closed DSLSH RXBIN")
set(root "${CMAKE_BINARY_DIR}/tools")
add_custom_target(the_native_tools ALL
 COMMAND ${CMAKE_COMMAND} -E remove_directory "${root}/sdk"
 COMMAND ${CMAKE_COMMAND} -E make_directory "${root}/sdk"
 COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:the_rxbin_host>" "${root}"
 COMMAND ${CMAKE_COMMAND} -E copy "${THE_CLASSIC_IMAGE}" "${root}/llcbrowser.rxbin"
 COMMAND ${CMAKE_COMMAND} -E copy "${THE_CREXX_HOST_BUILD}/bin/rxc${CMAKE_EXECUTABLE_SUFFIX}" "${THE_CREXX_HOST_BUILD}/bin/rxas${CMAKE_EXECUTABLE_SUFFIX}" "${THE_CREXX_HOST_BUILD}/bin/library.rxbin" "${THE_CREXX_HOST_BUILD}/bin/rxcexits.rxbin" "${root}/sdk"
 DEPENDS the_rxbin_host VERBATIM)
file(WRITE "${root}/cohort.json" "{\"version\":1,\"runtime\":\"${THE_CREXX_REVISION}\",\"classicImage\":\"${THE_CLASSIC_SHA256}\",\"classicSource\":\"7ac74bc6ced98a522222d5495bed51c1da38a077\"}\n")

if(THE_LANGUAGE_IMAGE)
 file(SHA256 "${THE_LANGUAGE_IMAGE}" language_sha)
 if(NOT language_sha STREQUAL "e549b76afa654e4a4264f8183de0c7435fe93688337eb13635c47a0c252f7e59")
  message(FATAL_ERROR "Language image differs from qualified checksum")
 endif()
 add_custom_command(TARGET the_native_tools POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "${THE_LANGUAGE_IMAGE}" "${root}/dslsh-classic.rxbin")
endif()
